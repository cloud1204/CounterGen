#include<bits/stdc++.h>
using namespace std;
#define int long long

void solve()
{
    int n,m;cin >> n >> m;
    vector<vector<int>> mp(n+1,vector<int>(m+1));
    vector<vector<vector<int>>> dp(n+1,vector<vector<int>>(m+1,vector<int>(2,-1e18)));
    for(int i=1;i<=n;i++) for(int j=1;j<=m;j++) cin >> mp[i][j];
    dp[1][1][0]=mp[1][1],dp[1][1][1]=-mp[1][1];
    for(int i=1;i<=n;i++)
    {
        for(int j=1;j<=m;j++)
        {
            if(i==1&&j==1) continue;
            dp[i][j][0]=max(dp[i-1][j][0],dp[i][j-1][0])+mp[i][j];
            dp[i][j][1]=min({max(dp[i-1][j][0],dp[i][j-1][1])+mp[i][j],max(dp[i-1][j][1],dp[i][j-1][0])+mp[i][j],max(dp[i-1][j][0],dp[i][j-1][0])-mp[i][j]});
        }
    }
    cout << dp[n][m][1] << endl;
}

signed main(void)
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    int T=1;cin >> T;
    while(T--) solve();
    return 0;
}
