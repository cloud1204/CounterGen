t = int(input())  
for _ in range(t):
    n, m = map(int, input().split())  
    grid = []
    for _ in range(n):
        row = list(map(int, input().split()))
        grid.append(row)
    r=sum(grid[0])
    i=1
    while(i<n):
        r+=grid[i][m-1]
        i+=1
    c=sum(grid[n-1])
    i=n-2
    while i>=0:
        c+=grid[i][0]
        i-=1
    print(min(r,c))
