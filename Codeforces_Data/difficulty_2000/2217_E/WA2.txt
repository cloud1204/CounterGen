#include<bits/stdc++.h>

using namespace std;
int check(long long n,int b,int p){
	while(n){
		long long r=n%b;
		n/=b;
		for(int i = 1;i<p;i++){
			if(n%b!=r)return 0;
			n/=b;
		}
	}
	return 1;
}
void solve(){
	long long n;cin>>n;
	int k=sqrt(n);
	int cnt=0;
	if(n==1||n==2){
		cout<<0<<"\n";
		return ;
	}
	for(int i=1;i<=k;i++){
		if(n%i==0){
			if(n%i-1<i)cnt++;
		}
	}
	for(int i=2;i<k;i++){
		for(int j=2;j<=pow(n,1*1.0/j);j++){
			if(check(n,i,j))cnt++;
		}
	}
	cout<<cnt<<"\n";
}
int main(){
	ios::sync_with_stdio(false);
	cin.tie(0);cout.tie(0);
	int t;
	cin>>t;
	while(t--)solve();
}
