#include<bits/stdc++.h>
using namespace std;
const int mod = 998244353;
int T,n,num,ans,mi,sum,tp[300005],nxt[300005];char s;
signed main(){
	scanf("%d",&T);
	while(T--){
		scanf("%d",&n);getchar();
		mi = 1,ans = num = sum = 0;
		for(int i = 1;i <= n;i++){
			scanf("%c",&s);
			s == '(' ? (ans += mi) %= mod,sum++ : sum--;
			nxt[i] = sum >= 2 ? nxt[i - 1] : i;
			mi = (mi << 1) % mod;
			tp[i] = tp[i - 1];
			s == '(' ? (tp[i] += num + 1 + tp[i - 1] - tp[nxt[i - 1]]) %= mod : (num += num + 1 + tp[i - 1] - tp[nxt[i - 1]]) %= mod;
		}
		printf("%d\n",(num + ans + mod) % mod);
	}
}
