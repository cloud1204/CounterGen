#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const ll mod = 998244353;

ll pw(ll x, ll y)
{
    ll ans=1;
    while (y>0)
    {
        if (y%2==1) ans=(ans*x)%mod;
        x=(x*x)%mod;
        y/=2;
    }
    return ans;
}

void solve()
{
    ll n;
    cin>>n;
    string s;
    cin>>s;
    ll ans=0, cnt=0;
    for (ll i=0;i<n;i++)
    {
        if (s[i]=='(') ans=(ans+pw(2,i))%mod;
    }
    ll good=1, bad=0, sum=0;
    for (ll i=0;i<n;i++)
    {
        if (s[i]==')') ans=(ans+good+bad)%mod;
        sum+=(s[i]=='('?1:-1);
        if (s[i]==')')
        {
            good=(2*good+bad)%mod;
            if (sum<2) bad=0;
        }
        else
        {
            if (sum>=2) bad=2*bad+good;
            else bad=0;
        }
    }
    cout<<ans<<"\n";
}

int main()
{
    ios_base::sync_with_stdio(false); cin.tie(0);
    ll tc;
    cin>>tc;
    while (tc--) solve();
}
