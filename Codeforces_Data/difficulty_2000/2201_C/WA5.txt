#include <bits/stdc++.h>

using namespace std;

typedef long long ll;

int const N = 3e5;

ll const mod = 998244353;

int n;
char str[N];
int s[N];
ll f[N][2];

void solve() {
    scanf("%d", &n);
    scanf("%s", str+1);

    ll power2 = 1;
    ll ans = 0;
    f[0][0] = 0;
    f[0][1] = 1;
    for(int i=1;i<=n;++i) {
        s[i] = s[i-1] + ((str[i] == '(' ) ? 1 : -1);
        if (str[i] == '(') {
            ans = (ans + power2) % mod;
            if (s[i]>=2) f[i][0] = (f[i-1][0] + f[i-1][0] + f[i-1][1]) % mod;
            else f[i][0] = 0;
            f[i][1] = f[i-1][1];
        }
        else {
            ans = (ans + f[i-1][0] + f[i-1][1]) % mod;
            if (s[i]>=2) f[i][0] = f[i-1][0];
            else f[i][0] = 0;
            f[i][1] = (f[i-1][1] + f[i-1][0] + f[i-1][1]) % mod;
        }
        power2 = (power2 + power2) % mod;
        // printf("#%d: %lld %lld, %lld\n", i, f[i][0], f[i][1], ans);
    }
    printf("%lld\n", ans);
}

int main() {
    // freopen("data", "r", stdin);
    int T;
    scanf("%d", &T);
    while (T--) solve();
    return 0;
}
