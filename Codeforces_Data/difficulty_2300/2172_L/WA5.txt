#include <bits/stdc++.h>
using namespace std;
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m, k;
    if (!(cin >> n >> m >> k)) return 0;
    string s;
    cin >> s;
    int seg = 1;
    for (int i = 1; i < n; ++i) if (s[i] != s[i-1]) ++seg;
    if (k == n) {
        cout << seg << "\n";
    } else {
        cout << min(n, seg + 2*m) << "\n";
    }
    return 0;
}
