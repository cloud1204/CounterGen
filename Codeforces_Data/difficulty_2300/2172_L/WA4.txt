def max_color_segments(n, m, k, rope):

    segments = 1
    for i in range(1, n):
        if rope[i] != rope[i - 1]:
            segments += 1
    

    max_additional = min(m * k, n - segments)
    

    return segments + max_additional


if __name__ == "__main__":
    n, m, k = map(int, input().split())
    rope = input().strip()
    print(max_color_segments(n, m, k, rope))
