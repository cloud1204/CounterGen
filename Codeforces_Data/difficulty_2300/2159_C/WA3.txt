#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int P = 1000000007;
void solve() {
	int n, cnt = 0; cin >> n;
	vector<ll> a(n + 10), dp(n + 10);
	for (int i = 0; i <= n; i++)
		cin >> a[i];
	if (n == 1) return cout << "1\n", void();
	for (int i = 1; i <= n; i++) {
		if (a[i] > 0 && a[a[i]] != -1 && a[a[i]] != i) return cout << "0\n", void();
		if (a[i] > 0) a[a[i]] = i;
	}
	for (int i = 1; i <= n; i++)
		cnt += (a[i] == -1);
	dp[0] = 1, dp[1] = 2;
	for (int i = 2; i <= cnt; i++)
		dp[i] = (dp[i - 1] * (1 + (i < cnt)) + dp[i - 2] * (i - 1)) % P;
	cout << dp[cnt] << '\n';
}
int main() {
	int T;
	cin >> T;
	while (T--)
		solve();
	return 0;
}
