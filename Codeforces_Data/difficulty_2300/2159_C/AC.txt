#include<bits/stdc++.h>
#define ll long long
#define p 1000000007
using namespace std;
ll T,n,a[400010],f[400010],fl,sm;
int main(){
	ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
	cin>>T;
	while(T--){
		cin>>n,fl=sm=0,f[0]=1,f[1]=2;
		for(int i=2;i<=n;i++)f[i]=(2*f[i-1]+(i-1)*f[i-2])%p;
		for(int i=0;i<=n;i++)cin>>a[i];
		for(int i=1;i<=n;i++)if(a[i]>0){
			if(a[i]>n||a[a[i]]>=0&&a[a[i]]!=i)fl=1;
			if(a[i]<=n)a[a[i]]=i;
		}
		for(int i=1;i<=n;i++)sm+=a[i]<0;
		cout<<(fl?0:(!sm?1:(f[sm]-f[sm-1]*(a[n]<0)+p)%p))<<"\n";
	}
	return 0;
}
