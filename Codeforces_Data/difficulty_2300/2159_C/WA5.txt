#include<bits/stdc++.h>
#define ll long long
#define endl '\n'
using namespace std;
const int P=1e9+7;
void work()
{
    int n;
    cin>>n;
    vector<int> a(n+1),b(n+1,0);
    for(int i=0;i<=n;++i) cin>>a[i];
    for(int i=1;i<=n;++i)
    {
    	if(a[i]!=-1&&a[a[i]]!=-1&&a[i]!=a[a[i]])
    	{
    		cout<<0<<endl;
    		return;
		}
		if(a[i]!=-1&&a[a[i]]==-1) b[i]=b[a[i]]=1;
	}
	int m=0;
	for(int i=1;i<=n;++i) m+=(b[i]==0);
	vector<int> f(m+1);
	f[0]=1;
	f[1]=2;
	for(int i=2;i<=m;++i) f[i]=(1ll*f[i-2]*(i-1)%P+f[i-1]*2%P)%P;
	if(b[n]) cout<<f[m]<<endl;
	else cout<<(f[m]-f[m-1]+P)%P<<endl;
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    int T=1;
    cin>>T;
    while(T--) work();
    return 0;
}
