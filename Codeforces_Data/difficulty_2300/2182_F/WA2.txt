#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using vll = vector<ll>;
using vvll = vector<vll>;
using pll = pair<ll,ll>;

const ll M=998244353;
vll v;
ll x;
vvll memo;
ll solve(ll i, ll d, ll mask){
    if(i==-1 || v[i] < d) return 0;
    if(memo[i][d]!=-1){
        return memo[i][d];
    }
    ll y=x&mask;
    ll r=0;
    ll bit=1ll<<(v[i]-d);
    if(bit >= y){
        r=(r+(1ll<<i))%M;
    }
    else if(bit << 1 > y){
        r = solve(i-1,d+1,mask&~bit);
    }
    return memo[i][d]=(r+solve(i-1,d,mask))%M;
}

int main(){
    ios::sync_with_stdio(0); cin.tie(0); cin.exceptions(cin.failbit);

    ll n,m;
    cin >> n >> m;
    v=vll(n);
    for(ll i=0;i<n;++i) cin >> v[i];
    sort(v.begin(), v.end());
    for(ll Q=0;Q<m;++Q){
        memo=vvll(n, vll(n,-1));
        ll cmd;
        cin >> cmd >> x;
        if(cmd==1){
            v.insert(lower_bound(v.begin(), v.end(), x),x);
            ++n;
        }
        else if (cmd==2){
            v.erase(find(v.begin(), v.end(), x));
            --n;
        }
        else{
            cout << solve(v.size()-1,0,-1) << endl;
        }
    }
}
