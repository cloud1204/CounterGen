vector<int> spf((int)1e6+ 1, 1);

// Calculating SPF (Smallest Prime Factor) for every number till MAXN.
void sieve()
{
    // stores smallest prime factor for every number
    spf[0] = 0;
    for (int i = 2; i <= (int)1e6; i++) {
        if (spf[i] == 1) { 
            
            // if the number is prime ,mark
            // all its multiples who havent
            // gotten their spf yet
            for (int j = i; j <= (int)1e6; j += i) {
                if (spf[j]== 1) 
                
                    // if its smallest prime factor is
                    // 1 means its spf hasnt been
                    // found yet so change it to i
                    spf[j] = i;
            }
        }
    }
}
ll modMultiply(ll a, ll b) {return ((ll) a * b) % MOD; }

ll modPow(ll a, ll e) {
    ll res = 1;
    a %= MOD;

    while (e > 0) {
        if (e & 1) res = modMultiply(res, a) % MOD;
        a = modMultiply(a, a) % MOD;
        e >>= 1;
    }
    return res;
}

ll modInverse(int a) {
    return modPow(a, MOD - 2);
}
vll fact;
void computeFactorials(int fSize) {
    fact.resize(fSize + 1, 1);
    fact[0] = 1;
    cf(i, 1, fSize) 
        fact[i] = modMultiply(fact[i-1], i);
}

vector<ll> invInt; vll invfact; 



void solve()
{
    int N, x; cin >> N >> x;

    vector<ll> multiply;
    vector<ll> addition;


    for (int i = 0; i < N; i++)
    {
        string op; cin >> op;
        if (op[0] == 'x') 
            multiply.pb(stoi(op.substr(1)));
        if (op[0] == '+') 
            addition.pb(stoi(op.substr(1)));
        if (op[0] == '/') 
            multiply.pb(modInverse(stoi(op.substr(1))));
        if (op[0] == '-')
            addition.pb(MOD-stoi(op.substr(1)));
    }

    int M = multiply.size();
    vector<ll> dp(M + 1);

    dp[0] = 1; // subset size 0
    for (auto &b : multiply)
    {
        for (int i = M; i >= 1; i--)
        {
            dp[i] = dp[i] + modMultiply(b, dp[i - 1]);
        }
    }

    for (int i = 0; i < M; i++)
    {
        //todos
        x = modMultiply(x, multiply[i]);
    }
    ll expect = 0;
    for (int i = 0; i <= M; i++)
    {
        //ev at i
        //linearity of expectation
        //ev0 ... evN
        expect = (expect + modMultiply(modMultiply(1, modInverse(M + 1)), modMultiply(dp[i], modMultiply(fact[i], modMultiply(fact[M - i], invfact[M]))))) % MOD;
        // i / m + 1 (chance to have this k subset of mutlipliers)
        //average value of k subset of mutlipliers
    }
    ll sum = 0;
    for (int i = 0; i < addition.size(); i++)
    {
        sum = (modMultiply(addition[i], expect) + sum) % MOD;
    }
    cout << (x + sum) % MOD << "\n";

}
