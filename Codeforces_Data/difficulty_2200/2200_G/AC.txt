for _ in range(int(input())):
    n,x=map(int,input().split())
    l=list(input().split())
    a1=[];a2=[];mod=1000000007
    for i in l:
        if i[0]=='x':
            a1.append(int(i[1:]))
        if i[0]=='/':
            a1.append(pow(int(i[1:]),-1,mod))
        if i[0]=='+':
            a2.append(int(i[1:]))
        if i[0]=='-':
            a2.append(-1*int(i[1:]))
    m=len(a1)
    f=[1]*(m+1)
    for i in range(1,m+1):f[i]=(f[i-1]*i)%mod
    dp=[[0]*(m+1) for i in range(m+1)]
    for i in range(m+1):
        dp[i][0]=1
    for i in range(1,m+1):
        for j in range(1,m+1):
            dp[i][j]=(dp[i-1][j]+a1[i-1]*dp[i-1][j-1])%mod
    s=0
    for i in range(m+1):
        den=(f[m]*pow(f[i],-1,mod)*pow(f[m-i],-1,mod))%mod
        s=(s+dp[m][i]*pow(den,-1,mod))%mod
    s=(s*pow(m+1,-1,mod))%mod
    print((sum(a2)*s+dp[m][m]*x)%mod)
