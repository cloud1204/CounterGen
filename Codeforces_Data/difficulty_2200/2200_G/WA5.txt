#include <bits/stdc++.h>
#define int long long
using namespace std;

const int mod = 1000000007;
int t, n, x, a[3010], ida, b[3010], idb;
int fct[3010], inv[3010], f[3010][3010];
string s;

int qpow(int x, int y) {
	int res = 1;
	while(y) {
		if(y&1) res = res*x%mod;
		x = x*x%mod, y >>= 1;
	}
	return res;
}
int C(int x, int y) {
	return (fct[x]*inv[y]%mod)*inv[x-y]%mod;
}
void tonum() {
	int num = 0;
	for(int i = 1; i < s.size(); i++)
		num = num*10+(s[i]-'0');
	if(s[0] == '+') a[++ida] = num;
	if(s[0] == '-') a[++ida] = -num;
	if(s[0] == 'x') b[++idb] = num;
	if(s[0] == '/') b[++idb] = qpow(num, mod-2);
}

signed main() {
	scanf("%lld", &t), fct[0] = inv[0] = 1;
	for(int i = 1; i <= 3000; i++) 
		fct[i] = fct[i-1]*i%mod, inv[i] = qpow(fct[i], mod-2);
	while(t--) {
		scanf("%lld%lld", &n, &x), ida = idb = 0;
		for(int i = 1; i <= n; i++) cin >> s, tonum();
		f[0][0] = 1;
		for(int i = 1; i <= idb; i++)
			for(int j = 0; j <= i; j++) {
				if(j) f[i][j] = (f[i-1][j]+f[i-1][j-1]*b[i]%mod)%mod;
				else f[i][j] = f[i-1][j]%mod;
			}
		/*for(int i = 1; i <= idb; i++) {
			for(int j = 0; j <= i; j++)
				cout << f[i][j] << ' ';
			cout << endl;
		}*/
		int s = C(n, idb+1)*fct[ida-1]%mod, cnt = 0, ml = 1;
		for(int i = 1; i <= idb; i++) ml = ml*b[i]%mod;
		for(int i = 1; i <= ida; i++)
			for(int j = 0; j <= idb; j++)
				cnt += ((a[i]*f[idb][j]%mod)*fct[j]%mod)*fct[idb-j]%mod;
		s = (s*cnt%mod)*inv[n]%mod, s = (s+x*ml%mod)%mod;
		printf("%lld\n", (s+mod)%mod);
	}
	
	return 0;
}
