#include<iostream>
using namespace std;
using ll = long long;

const int N = 5e5;
const ll M = 998244353;

ll inv(ll a){
	ll m=M,y=0,x=1;
	while(a>1){
		ll q=a/m,t=m;
		m=a%m,a=t,t=y;
		y=x-q*y,x=t;
	}
	if(x<0) x+=M;
	return x;
}

ll quo(ll p,ll q){
	return (p*inv(q))%M;
}

int main(){
	int n,m; cin>>n>>m;
	ll a[N];
	for(int i=0;i<n;i++)
		cin>>a[i];
	for(int x=0;x<m;x++){
		int k; cin>>k;
		ll ans=0;
		for(int i=0;i<n;i++){
			int l=i,r=i;
			while(l>0&&a[l-1]>a[i]) l--;
			while(r<n-1&&a[r+1]>=a[i]) r++;
			for(int j=l;j<i;j++)
				ans=(ans+quo((j-l+1)*(r-i+1),a[j]))%M;
			for(int j=i+1;j<=r;j++)
				ans=(ans+quo((i-l+1)*(r-j+1),a[j]))%M;
			if(k<a[i]-1)
				ans=(ans+quo((i-l+1)*(r-i+1)*(k+1),a[i]))%M;
			else
				ans=(ans+(i-l+1)*(r-i+1)*(k-a[i]+2))%M;
		}
		cout<<ans<<endl;
	}
}
