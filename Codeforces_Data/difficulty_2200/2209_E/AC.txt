#include <bits/stdc++.h>
using namespace std;
int T;
char s[1000005];
int f[1000005],dp[1000005];
int l,r,q,n;
int main(){
	cin>>T;
	while(T--){
		scanf("%d%d%s",&n,&q,s+1);
		while(q--){
			scanf("%d%d",&l,&r);
			long long ans=0;
			for(int i=l;i<=r;i++){
				f[i-l+1]=f[i-l];
				while(f[i-l+1]&&s[f[i-l+1]+l]!=s[i]) f[i-l+1]=f[f[i-l+1]];
				if(s[f[i-l+1]+l]==s[i]&&f[i-l+1]+l!=i) f[i-l+1]++;
				dp[i-l+1]=1;
				dp[i-l+1]=dp[f[i-l+1]]+dp[i-f[i-l+1]-l+1];
				ans+=dp[i-l+1];
			}
			printf("%lld\n",ans);
		}
	}
	return 0;
}
