#include<bits/stdc++.h>
using namespace std;
const int N=1e6+7;
int T,n,q;
string s;
void solve(){
	cin>>n>>q;
	cin>>s;
	s=" "+s;
	while(q--){
		int x,y;
		cin>>x>>y;
		int l=1,r=x+1;
		int cnt=1,ans=1;
//		cout<<"--"<<" "<<q<<"\n";
		while(r<=y){
			if(s[r]==s[x]) cnt++,l=1;
			else if(s[r]!=s[x+l]) {
				l=1;
				cnt=1;
			}else{
				l++;
			}
			ans+=cnt;
//			cout<<r<<" "<<cnt<<"\n";
			r++;
		}
		cout<<ans<<"\n";
	}
}
int main(){
	cin>>T;
	while(T--){
		solve();
	}
}
