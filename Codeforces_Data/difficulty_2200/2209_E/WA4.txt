#include <bits/stdc++.h>
using namespace std;
template <class T>
using vec = vector<T>;
using ll = long long;
using pii = pair<int, int>;
using vi = vec<int>;

#define pb push_back
#define all(x) begin(x), end(x)
#define sz(x) (int)(x).size()
#define rep(i,a,b) for (int i = a; i < (b); ++i)

void solve() {
	int n, q; cin >> n >> q;
	string s; cin >> s;
	while (q--) {
		ll l, r; cin >> l >> r;
		l--;
		ll hans = 0, ans = 0;
		int ind = l;
		rep(i,l,r) {
			if (s[i] == s[l]) {
				hans++;
				ind = l+1;
			} else if (s[ind] == s[i]) {
				ind++;
			} else {
				hans = 1;
			}
			ans += hans;
		}
		cout << ans << '\n';
	}
}

int main() {
	cin.tie(0)->sync_with_stdio(0);
	cin.exceptions(cin.failbit);
	int t; cin >> t;
	while (t--) solve();
}
