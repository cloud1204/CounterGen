#include <bits/stdc++.h>
using namespace std;
using ll = long long;

const int MOD = 998244353;
const int MAXN = 200005;

vector<int> tree[MAXN];
int sizes[MAXN];

void dfs(int v, int p) {
    sizes[v] = 1;
    for (int u : tree[v]) {
        if (u == p) continue;
        dfs(u, v);
        sizes[v] += sizes[u];
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t; cin >> t;
    while (t--) {
        int n; cin >> n;
        for (int i = 1; i <= n; ++i) tree[i].clear();

        for (int i = 0; i < n-1; ++i) {
            int u, v; cin >> u >> v;
            tree[u].push_back(v);
            tree[v].push_back(u);
        }

        dfs(1, 0);

        int white_count = 0;
        for (int i = 1; i <= n; ++i)
            if (sizes[i] % 2 == 0) white_count++;

        ll ans = 1;
        if (white_count > 0) {
            ans = 1;
            for (int i = 0; i < white_count-1; ++i)
                ans = (ans * 2) % MOD;
        }

        cout << ans << "\n";
    }
}
