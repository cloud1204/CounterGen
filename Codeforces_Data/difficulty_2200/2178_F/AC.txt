#include<bits/stdc++.h>
#define ll long long
#define eb emplace_back
using namespace std;
const ll _=2e5+5,M=998244353;
ll N,n,m,x,y,a[_],b[_],V[_],F[_],i;vector<ll>E[_];
void p(ll x,ll fa){
	a[x]=1;
	for(ll y:E[x])if(y!=fa)p(y,x),a[x]+=a[y];
	if(x!=1&&!(a[x]&1))b[++m]=a[x],a[x]=0;
}
void P(){
	cin>>n;m=0;
	for(i=0;i<=n;i++)a[i]=b[i]=0,E[i].clear();
	for(i=1;i<n;i++)cin>>x>>y,E[x].eb(y),E[y].eb(x);
	p(1,0);b[0]=a[1];
//cout<<"### "<<m<<"    ";
//for(i=0;i<=m;i++)cout<<b[i]<<' ';cout<<'\n';
	if(m==0){cout<<"1\n";return;}
	for(y=0,x=i=1;i<=m;i++)x=x*b[i]%M,y+=V[b[i]];
	cout<<x*x%M*b[0]%M*(y%M)%M*F[m-1]%M<<'\n';
}
int main(){
	ios::sync_with_stdio(0);cin.tie(0);cout.tie(0);
	for(V[1]=1,i=2;i<_;i++)V[i]=(M-M/i)*V[M%i]%M;
	for(F[0]=i=1;i<_;i++)F[i]=F[i-1]*i%M;
	cin>>N;while(N--)P();
}
