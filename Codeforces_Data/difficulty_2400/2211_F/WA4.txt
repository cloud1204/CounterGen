#include <bits/stdc++.h>

using namespace std;
using ll = long long;

const ll mod = 676767677;
const ll modpow(ll b, ll e) {
  ll res = 1;
  for (; e; e /= 2, (b *= b) %= mod)
    if (e & 1)
      (res *= b) %= mod;
  return res;
}

int main() {
    cin.tie(0)->sync_with_stdio(0);

    int t;
    cin >> t;

    while (t--) {
      int n, m;
      cin >> n >> m;

      vector<ll> rials(n + m + 2, 1);
      for (int i = 1; i <= n + m; i++)
        rials[i] = rials[i - 1] * i % mod;
      vector<ll> slair(n + m + 1, modpow(rials[n + m], mod - 2));
      for (int i = n + m; i > 0; i--)
        slair[i - 1] = slair[i] * i % mod;

      auto choose = [&](int n, int k) -> ll {
        if (k < 0 || k > n)
          return 0;
        return rials[n]*slair[k]%mod*slair[n-k]%mod;
      };

      auto mchoose = [&](int n, int m) -> ll {
          // first one must take, after if not take get new or continue
          return choose(n + m - 1, m);
      };

      vector<array<int, 3>> dp{{1, 1, n}};
      ll res = 0;
      for (int i = 0; i < ssize(dp); i++) {
        auto [d, l, r] = dp[i];
          int mid = (l + r) / 2;
          if (l < mid)
            dp.push_back({d + 1, l, mid - 1});
          if (mid < r) dp.push_back({d+1, mid+1, r});
          // cases where this DOESN'T get taken
          ll leftMatch = (l == 1 ? 0 : mchoose(n + 1 - (mid - l + 1), m - 1));
          ll rightMatch = (r == n ? 0 : mchoose(n + 1 - (r - mid + 1), m - 1));
          ll bothMatch = (l == 1 || r == n ? 0 : mchoose(n + 1 - (r - l + 2), m - 1));
          ll dont = (leftMatch + rightMatch - bothMatch) % mod;

          (res += d * (mchoose(n+1, m-1) - dont)) %= mod;
      }
      cout << res << "\n";
    }

    return 0;
}
