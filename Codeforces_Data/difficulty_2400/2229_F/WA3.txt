#include<bits/stdc++.h>
#define int long long
using namespace std;
int rd(){// {{{
	int r=0;bool w=0;char ch=getchar();
	while(ch<'0'||ch>'9')  w=ch=='-'?1:0,ch=getchar();
	while(ch>='0'&&ch<='9')  r=r*10+(ch^48),ch=getchar();
	return w?-r:r;
}// }}}
const int N=19;
int n,k,a[N];
pair<int,int> f[1<<N];
bool check(int lim){
	int m=n,res=0;
	for(;m&&a[m]>=lim;m--)  ++res;
	for(int i=0;i<(1<<m);i++)  f[i]={0,0};
	for(int i=0;i<(1<<m)-1;i++)for(int j=1;j<=m;j++)if(!((i>>j-1)&1))
		if(f[i].second+a[j]>=lim)
			f[i|(1<<j-1)]=max(f[i|(1<<j-1)],{f[i].first+1,0});
		else  f[i|(1<<j-1)]=max(f[i|(1<<j-1)],{f[i].first,f[i].second+a[j]});
	return res+f[(1<<m)-1].first>=k;
}
signed main(){
for(int T=rd();T--;){
	n=rd(),k=rd();
	for(int i=1;i<=n;i++)  a[i]=rd();
	sort(a+1,a+n+1);--n;
	int l=1,r=1e18;
	while(l<r){
		int mid=l+r+1>>1;
		if(check(mid))  l=mid;
		else  r=mid-1;
	}
	printf("%lld\n",l+a[n+1]);
}
	return 0;
}
