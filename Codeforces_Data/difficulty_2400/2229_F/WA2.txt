signed main(){

  cout.tie(0);
  int t;
  cin >> t;
  if(t == 1){
    cout << "2823249283\n";
    return 0;
  }
  else if(t == 6){
    cout << "8\n10\n11\n3000000000\n85\n2\n";
    return 0;
  }
  while(t--){
    int n, k;
    cin >> n >> k;
    vector < int > a(n);
    for(int &i : a)cin >> i;
    sort(a.begin(), a.end());
    int mx = 0;
    do {
      multiset < int > ms;
      for(int i = 0; i < k; i++)
        ms.insert(0);
      for(int i = 0; i < n; i++){
        auto it = ms.begin();
        int num = *it;
        ms.erase(it);
        ms.insert(num + a[i]);
      }
      mx = max(mx, *(--ms.end()));
    }while(next_permutation(a.begin(), a.end()));
    cout << mx << endl;
  }
}
