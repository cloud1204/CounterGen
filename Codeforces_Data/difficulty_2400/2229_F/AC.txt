#include<bits/stdc++.h>
#define ll long long
#define fi first
#define se second
using namespace std;
ll T,n,k,l,r,m,ans,a[20];
pair<ll,ll>f[1<<18],x;
int main(){
	ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
	for(cin>>T;T--;){
		cin>>n>>k,l=r=ans=0;
		for(ll i=0;i<n;i++)cin>>a[i],r+=a[i];
		for(sort(a,a+n),r-=a[--n];l<=r;){
			m=l+r>>1;
			for(ll i=1;i<1<<n;i++){
				f[i]={0,0};
				for(ll j=0;j<n;j++)if(i>>j&1){
					x=f[i^1<<j],x.se+=a[j];
					if(x.se>=m)x.fi++,x.se=0;
					f[i]=max(f[i],x);
				}
			}
			if(f[(1<<n)-1].fi>=k)ans=m,l=m+1;
			else r=m-1;
		}cout<<ans+a[n]<<"\n";
	}
	return 0;
}
