#include<bits/stdc++.h>
#define int long long
using namespace std;
const int MAXN = 1e6 + 5;
int n, k;
int a[MAXN];
pair<int, int> dp[1 << 17];
bool check(int x){
    for(int i = 0;i < (1 << (n - 1));i ++){
        dp[i] = {0, 0};
        for(int j = 0;j < n - 1;j ++){
            if(((i >> j) & 1) == 0)continue;
            auto [ok, val] = dp[i ^ (1 << j)];
            val += a[j];
            if(val >= x)ok ++, val = 0;
            dp[i] = max(dp[i], {ok, val});
        }
    }
    return dp[(1 << (n - 1)) - 1].first >= k;
}
int find(int l, int r){
    if(l == r)return l;
    int mid = (l + r + 1) >> 1;
    if(check(mid))return find(mid, r);
    else return find(l, mid - 1);
}
void solve(){
    cin >> n >> k;
    for(int i = 0;i < n;i ++)cin >> a[i];
    sort(a, a + n);
    if(n == 1)cout << a[1] << "\n";
    else cout << find(0, 1e18) + a[n - 1] << "\n";
    return ;
}
signed main(){
    ios::sync_with_stdio(false);cin.tie(0);cout.tie(0);
    int t = 1;
    cin >> t;
    while(t --){
        solve();
    }
    return 0;
}
