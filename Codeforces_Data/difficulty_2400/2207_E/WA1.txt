#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define ar array
const int mod = 1e9+7; 
void solve(){
    int n; 
    cin >> n; 
    vector<int> v(n);
    for (int &i:v) cin >> i; 
    for (int i = 0; i < n; i++) if (v[i]<n-i-1||v[i]>n||i&&v[i]>v[i-1]) {cout << "0\n"; return;}
    int ans=1; 
    for (int i = 0; i < n; i++) {
        if (i&&v[i]==v[i-1]) ans=(ans*(v[i]-(n-i-1)))%mod;
        else ans=(ans*(i+1))%mod; 
    }
    cout << ans << '\n';
}
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int tt;
    cin >> tt;
    while(tt--) {
        solve();
    }
    return 0;
}
