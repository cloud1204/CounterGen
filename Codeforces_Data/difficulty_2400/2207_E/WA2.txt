#include <bits/stdc++.h>
using namespace std;
#define ED { cout << 0 << '\n'; return; }
const int mod = 1e9 + 7;
int a[400005];
void solve() {
  int n; cin >> n;
  for (int i = 0; i < n; ++i) cin >> a[i];
  for (int i = 0; i < n - 1; ++i) if (a[i] < a[i + 1]) ED
  for (int i = 0; i < n; ++i) if (a[i] < n - i - 1) ED
  if (a[0] > n || a[0] < n - 1) ED
  int ans = a[0] == n ? n : 1;
  for (int i = 1; i < n; ++i)
    if (a[i] == a[i - 1] || a[i - 1] > a[i] + 1)
      ans = 1LL * ans * (n - i) % mod;
  cout << ans << '\n';
}
signed main() {
  ios::sync_with_stdio(0); cin.tie(0);
  int _; cin >> _; while (_--) solve();
  return 0;
}
