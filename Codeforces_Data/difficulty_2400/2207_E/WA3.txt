#include<bits/stdc++.h>
#define int long long

using namespace std;


void solve(){
	int n;
	cin >> n;
	vector<int> a(n+1);
	for(int i = 1; i <= n; i ++){
		cin >> a[i];	
	}
	
	a[0] = n;
	for(int i = 1; i <= n; i ++){
		if(a[i] > a[i-1] || a[i] > n || a[i] < n - i){
			cout << 0 << '\n';
			return;
		}
	}
	int cnt = a[n];
	int ans = 1;
	for(int i = n - 1; i >= 1; i --){
		if(a[i] == a[i+1]){
			ans *= cnt;
			cnt --;
		}else{
			cnt += a[i] - a[i+1] - 1;
			ans *= n - a[i] + 1 + cnt;
		}
	}
	cout << ans << '\n';
}

signed main(){
	ios::sync_with_stdio(false);
	cin.tie(0);
	
	int t = 1;
	cin >> t;
	while(t --){
		solve();
	}
}
