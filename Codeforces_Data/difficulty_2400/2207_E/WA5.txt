#include <bits/stdc++.h>
#define int long long
#define MOD 1000000007
#define endl '\n'
using namespace std;
int T,n,k,st,a1,a2;
int a[500011],bl[500011],l,ans,cur,num,chs;
bool b[500011],ck;
signed main(){
	ios::sync_with_stdio(0);
	cin.tie(0);
	cout.tie(0);
	cin>>T;
	while(T--){
		cin>>n;
		for(int i=1;i<=n;i++){
			cin>>a[i];
		}
		a[0]=n;
		ans=1;
		for(int i=1;i<=n;i++){
			if(a[i]>a[i-1]||a[1]-a[i]>i-1||a[1]>n||a[1]<n-1){
				ans=0;
				break;
			}
		}
		if(ans==0){
			cout<<0<<endl;
			continue;
		}
		if(a[1]==n)num=1;
		else num=0;
		chs=0;
		for(int i=2;i<=n;i++){
			if(a[i]==a[i-1]){
				num++;
			}else{
				ans*=(num+n-a[i-1]+1-chs);
				ans%=MOD;
				//cout<<(num+n-a[i-1]+1)<<" "<<i<<endl;
				for(int j=a[i-1]-1;j>a[i];j--){
					ans*=num-chs;
					ans%=MOD;
					chs++;
				}
			}
		}
		for(int i=a[n]-1;i>=0;i--){
			ans*=(i+1);
			ans%=MOD;
		}
		cout<<ans<<endl;
	}
}
