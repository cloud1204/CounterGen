#include <bits/stdc++.h>
using namespace std;
using C = complex<double>;
const double pi = acos(-1);

namespace FFT {
    int A[500009], B[500009];
    
    void fft(vector<C> &A, bool inv) {
    	int N = A.size(), i, j, l, b;
    	for(i=1, j=0; i<N; i++) {
    		for(b=N>>1; j>=b; b>>=1) j -= b;
    		j += b;
    		if(i < j) swap(A[i], A[j]);
    	}
    	for(l=1; l<N; l<<=1) {
    		C wl = polar(1., pi/l*(inv?-1:1));
    		for(i=0; i<N; i+=2*l) {
    			C w(1);
    			for(j=0; j<l; j++) {
    				C a = A[i|j], b = A[i|j|l]*w;
    				A[i|j] = a+b; A[i|j|l] = a-b;
    				w *= wl;
    			}
    		}
    	}
    	if(inv) for(int i=0; i<N; i++) A[i] /= N;
    }
    
    vector<int> mul(vector<int> A, vector<int> B) {
    	vector<C> FA(A.begin(), A.end()), FB(B.begin(), B.end());
    	int N = A.size(), M = B.size(), sz = 1;
    	while(sz < N || sz < M) sz *= 2;
    	sz *= 2;
    	FA.resize(sz); FB.resize(sz);
    	fft(FA, 0); fft(FB, 0);
    	for(int i=0; i<sz; i++) FA[i] *= FB[i];
    	fft(FA, 1);
    	A.resize(sz);
    	for(int i=0; i<sz; i++) A[i] = int(FA[i].real() + (FA[i].real()>0?0.5:-0.5));
    	return A;
    }
}

int D[500009];

int main() {
    int N, K; scanf("%d%d", &N, &K);
    vector<int> A(N), B(N);
    for (int i=0; i<N; i++) {
        scanf("%1d", &A[i]);
        if (!A[i]) B[i] = 1;
    }
    reverse(B.begin(), B.end());
    vector<int> C = FFT::mul(A, B);
    for (int i=0; i<N-1; i++) D[N-i-1] = C[i];
    for (int i=N; i<2*N-1; i++) D[i-N+1] += C[i];
    int s = 0;
    for (int i=0; i<N; i++) if (A[i] == 0) s += (i+1 < N ? A[i+1] : 0) + (i ? A[i-1] : 0);
    for (int i=1; i<=N; i++) {
		D[i] += s;
		for (int j=i-1; j+1<N; j+=i) if (A[j] == 0 || A[j+1] == 0) D[i] -= A[j] + A[j+1];
	}
    sort(D+1, D+N+1);
    printf("%d", D[N-K+1]);
	return 0;
}
