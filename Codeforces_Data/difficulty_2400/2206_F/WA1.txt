#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

int n, k;
string s;

// Function to compute f(w) efficiently
ll compute_f(int w) {
    int rows = (n + w - 1) / w;
    vector<vector<int>> grid(rows, vector<int>(w, 0));
    
    // Fill the grid
    for (int i = 0; i < n; i++) {
        int row = i / w;
        int col = i % w;
        grid[row][col] = s[i] - '0';
    }
    
    ll sum = 0;
    // For each cell without mines, count adjacent mines
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < w; j++) {
            if (grid[i][j] > 0) continue;
            
            int mines = 0;
            if (i > 0) mines += grid[i-1][j];
            if (i < rows - 1) mines += grid[i+1][j];
            if (j > 0) mines += grid[i][j-1];
            if (j < w - 1) mines += grid[i][j+1];
            
            sum += mines;
        }
    }
    
    return sum;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    
    cin >> n >> k;
    cin >> s;
    
    // Since we need the k-th largest, we can use nth_element for partial sorting
    vector<ll> values(n);
    
    for (int w = 1; w <= n; w++) {
        values[w-1] = compute_f(w);
    }
    
    // Use nth_element to find the k-th largest without full sorting
    nth_element(values.begin(), values.begin() + k - 1, values.end(), greater<ll>());
    
    cout << values[k-1] << endl;
    
    return 0;
}
