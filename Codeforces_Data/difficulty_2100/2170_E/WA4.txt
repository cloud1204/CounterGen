#include <bits/stdc++.h>
using namespace std;
const int MOD=998244353;

int main(){
    int t;cin>>t;
    while(t--){
        int n,m;
        cin>>n>>m;
        int M=n-1;
        vector<int> best(M+1,0);
        for(int i=0;i<m;i++){
            int l,r;cin>>l>>r;
            if(r-1<=M)
                best[r-1]=max(best[r-1],l);
        }

        vector<long long> dp(M+1,0);
        dp[0]=1;
        for(int i=1;i<=M;i++){
            long long sub=0;
            if(best[i]>0)sub=dp[best[i]-1];
            dp[i]=(2*dp[i-1]%MOD-sub%MOD+MOD)%MOD;
        }
        cout<<(dp[M]*2%MOD)<<endl;
    }
}
