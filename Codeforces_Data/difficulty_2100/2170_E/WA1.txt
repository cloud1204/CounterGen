#include<bits/stdc++.h>
const int P=998244353;
using namespace std;
int main(){
	int t;
	cin>>t;
	while(t--){
		int n,m;
		cin>>n>>m;
		vector<int> l(m),r(m),bnd(n,0),dp(n);
		for(int i=0;i<m;i++){
			cin>>l[i]>>r[i];
			bnd[r[i]-1]=l[i];
		}
		dp[0]=2;
		for(int i=1;i<n;i++) dp[i]=(2*dp[i-1]-(bnd[i]?(dp[bnd[i]-1]):0))%P;
		cout<<dp[n-1]<<'\n';
	}
}
