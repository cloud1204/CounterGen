mod = 998244353

import sys
input = sys.stdin.readline

_ = int(input())
for _ in range(_):
    n, m = map(int, input().split())
    ms = [0] * n
    dp = [0] * n
    dp[0] = 1
    for _ in range(m):
        l, r = map(int, input().split())
        ms[r-1] = max(ms[r-1], l)
    for i in range(1, n):
        if ms[i] == 0:
            dp[i] = 2 * dp[i-1] % mod
        else:
            dp[i] = (2 * dp[i-1] - dp[ms[i]-1]) % mod
    print((2 * dp[-1]) % mod)
