mod=998244353
t=int(input())
for _ in range(t):
  n,m=map(int,input().split())
  l=[0]*(n+1)
  for _ in range(m):
    c,c1=map(int,input().split())
    l[c1]=max(l[c1],c)
  for i in range(1,n+1):
    l[i]=max(l[i],l[i-1])
  ok=[0]*(n+2)
  ok[1]=2
  for i in range(1,n+1):
    ok[i+1]=(ok[i]+(ok[i]-ok[l[i]]+mod)%mod)%mod
  res=(ok[n+1]-ok[n]+mod)%mod
  print(res)
