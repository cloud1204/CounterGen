import sys
input = sys.stdin.readline

mod=998244353

t=int(input())

for tests in range(t):
    n,m=map(int,input().split())

    LIST=[[] for i in range(n+3)]

    for i in range(m):
        l,r=map(int,input().split())
        LIST[l].append(r)

    DP0=[0]*(n+1)
    DP1=[0]*(n+1)
    DP0[0]=1
    for i in range(1,n+1):
        DP0[i]=(DP0[i]+DP0[i-1]*2)%mod
        DP1[i]=(DP1[i]+DP1[i-1]*2)%mod

        for to in LIST[i]:
            DP0[to]=(DP0[to]+DP1[i])%mod
            DP1[to]=(DP1[to]+DP0[i])%mod

    print((DP0[-1]-DP1[-1])%mod)
