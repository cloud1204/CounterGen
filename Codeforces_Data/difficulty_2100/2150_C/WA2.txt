#include<bits/stdc++.h>
using namespace std;
#define ll long long
const int N=2e3+10,off=(1<<18),INF=1e18;
ll n,v[N],a[N],b[N],ind[N],dp[N][N],tour[4*off+5];
ll f(int i,int j){
	if(i==n and j==n)return 0;
	if(dp[i][j]==-INF){
		if(i!=n){
			if(ind[a[i]]>=j)dp[i][j]=max(dp[i][j],v[a[i]]+f(i+1,j));
			else dp[i][j]=max(dp[i][j],f(i+1,j));
		}
		if(j!=n)dp[i][j]=max(dp[i][j],f(i,j+1));
	}
	return dp[i][j];
}
void solve(){
	cin>>n;
	for(int i=0;i<n;i++)cin>>v[i];
	for(int i=0;i<n;i++)cin>>a[i],a[i]--;
	for(int i=0;i<n;i++)cin>>b[i],b[i]--,ind[b[i]]=i;
	for(int i=0;i<n;i++)for(int j=0;j<n;j++)dp[i][j]=-INF;
	cout<<f(0,0)<<'\n';
	/*for(int i=0;i<n;i++){
		for(int j=0;j<n;j++)cout<<dp[i][j]<<' ';
		cout<<'\n';
	}*/
	return;
}
int main(){
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	int t;
	cin>>t;
	while(t--)solve();
	return 0;
}
