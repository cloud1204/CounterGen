#include <bits/stdc++.h>
#ifndef DEBUG
#define debug
#endif
using namespace std;
using i64 = int64_t;

void solve() {
  int n;
  cin >> n;
  vector<int> v(n + 1), a(n + 1), b(n + 1), pos(n + 1);
  for (int i = 1; i <= n; i ++) cin >> v[i];
  for (int i = 1; i <= n; i ++) cin >> a[i];
  for (int i = 1; i <= n; i ++) {
    cin >> b[i];
    pos[b[i]] = i;
  }
  vector<vector<i64>> dp(n + 2, vector<i64>(n + 2, LLONG_MIN / 2));
  dp[1][1] = 0;
  for (int i = 1; i <= n; i ++) {
    for (int j = 1; j <= n; j ++) {
      dp[i][j + 1] = max(dp[i][j + 1], dp[i][j]);
      if (pos[a[i]] >= j) {
        dp[i + 1][j] = max(dp[i + 1][j], dp[i][j] + v[a[i]]);
      } else {
        dp[i + 1][j] = max(dp[i + 1][j], dp[i][j]);
      }
    }
  }
  i64 ans = 0;
  for (int i = 1; i <= n; i ++) ans = max(ans, dp[n + 1][i]);
  cout << ans << '\n';
}
int main() {
  cin.tie(0)->sync_with_stdio(0);
  int T = 1;
  cin >> T;
  for (int cas = 1; cas <= T; cas ++) {
    solve();
  }
}
