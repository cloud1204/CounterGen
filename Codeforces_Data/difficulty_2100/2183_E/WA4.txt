def solve():
    n,m = [int(i) for i in input().split()]
    a = [int(i) for i in input().split()]
    dp = [[0 for j in range(1+m)] for i in range(n)]
    if a[0] > 1:
        print(0)
        return
    dp[0][1]=1
    for i in range(1,n):
        for j in range(1,m+1):
            for k in range(1,m//j):
                if a[i] == 0 or a[i] == j*k+j:
                    dp[i][j*k+j] += dp[i-1][j*k]
                    dp[i][j*k+j] %= 998244353
    print(sum(dp[-1]))
    
t = int(input())
for i in range(t):
    solve()
