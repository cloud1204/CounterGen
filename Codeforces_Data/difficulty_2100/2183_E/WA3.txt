import math
mod=998244353

def get(i,k,dic,m,L):
    if i==len(L):
        return 1
    if (i,k) in dic:
        return dic[(i,k)]
    ans=0
    if L[i]!=0:
        if math.gcd(L[i],k)==L[i]-k:
            ans=get(i+1,L[i],dic,m,L)
    else:
        for j in range (k+1,m):
            if math.gcd(j,k)==j-k:
                ans=(ans+get(i+1,j,dic,m,L))%mod
    dic[(i,k)]=ans
    return ans
            
t=int(input())
for _ in range (0,t):
    n,m=map(int,input().split(" "))
    L=list(map(int,input().split(" ")))
    if L[0]==0:
        L[0]=1
    if L[0]!=1:
        print(0)
        continue
    ans=get(1,1,{},m,L)
