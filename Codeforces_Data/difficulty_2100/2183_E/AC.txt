#include<bits/stdc++.h>

using namespace std;
typedef long long ll;
ll MOD=998244353;
int main(){
	int t;
	cin >> t;
	while(t--){
		ll n,m;
		cin>>n>>m;
		vector<ll> a(n+1,0);
		vector<vector<ll>> dp(n+1,vector<ll>(m+1,0));
		for(int i=1;i<=n;i++)cin>>a[i];
		if(a[1]<2)dp[1][1]=1;
		for(int i=2;i<=n;i++)for(int j=1;j<=m;j++)for(int k=j+j;k<=m;k+=j)if(!a[i]||a[i]==k)dp[i][k]=(dp[i][k]+dp[i-1][k-j])%MOD;
		ll ans=0;
		for(int i=n;i<=m;i++)ans=(ans+dp[n][i])%MOD;
		cout<<ans<<"\n";
	}
}
