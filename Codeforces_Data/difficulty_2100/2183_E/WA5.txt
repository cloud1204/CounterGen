#include<bits/stdc++.h>
#define ll long long
using namespace std;
ll M=998244353,N=3005;
int main(){
    ios_base::sync_with_stdio(0); cin.tie(0);
    ll siuu=1;cin>>siuu;
    vector<ll> fac[N];
    for(ll i=1;i<N;i++){
        for(ll j=i;j<N;j+=i)fac[j].push_back(i);
    }
    while(siuu--){
        ll n,m;cin>>n>>m;
        ll a[n],ans=0;
        for(ll i=0;i<n;i++)cin>>a[i];
        vector<vector<ll> > dp(n,vector<ll> (m+1,0));
        dp[0][1]=1;
        for(ll i=1;i<n;i++){
            for(ll j=1;j<m+1;j++){
                for(auto it:fac[j]){
                    if(j+it>m)break;
                    dp[i][j+it]=(dp[i][j+it]+dp[i-1][j])%M;
                }
                if(a[i]!=0){
                    if(a[i]!=j)dp[i][j]=0;
                }
            }
        }
        for(ll j=1;j<m+1;j++)ans=(ans+dp[n-1][j])%M;
        cout<<ans<<endl;
    }
}
