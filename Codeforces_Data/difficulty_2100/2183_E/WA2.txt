#include<bits/stdc++.h>
#define int long long
#define mod 998244353
#define N 3001
using namespace std;

int n,m,a[N],f[N][N];
vector<int>d[N];
signed main(){
	for(int i=1;i<=N-1;++i)for(int j=i;j<=N-15;j+=i)d[j].push_back(i);
	int t;cin>>t;while(t--){
		cin>>n>>m;for(int i=1;i<=n;i++)cin>>a[i];
		for(int i=1;i<=n;i++)for(int j=1;j<=m;j++)f[i][j]=0;
		a[n+1]=a[1];
		if(a[1]>1){cout<<0<<endl;continue;}f[1][1]=1;
		for(int i=2;i<=n+1;i++){
			for(int j=1;j<=m;j++){
				if(a[i]&&a[i]!=j)continue;
				for(int x:d[j])f[i][j]=(f[i][j]+f[i-1][j-x])%mod;
			}
		}int ans=0;for(int j=1;j<=m;++j)ans=(ans+f[n][j])%mod;
		cout<<ans<<endl;
	}
	return 0;
}
