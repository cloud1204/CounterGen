#include<bits/stdc++.h>
using namespace std;
const int N=8005,mod=998244353;
int n,a[N],f[N],g[N][N];
signed main(){
int T;scanf("%d",&T);while(T--){
	scanf("%d",&n);
	for(int i=1;i<=n;i++)  scanf("%d",a+i);
	for(int l=1;l<=n;l++){
		for(int r=l+1,j=0;r<=n;r++){
			while(j&&a[r]!=a[j+l])  j=g[l][j];
			if(a[r]==a[j+l])  j++;
			g[l][r]=j;
		}
	}
	f[0]=1;
	for(int r=1;r<=n;r++)for(int l=0;l<r;l++)if(!g[l+1][r])  f[r]=(f[r]+f[l])%mod;
	printf("%d\n",f[n]);
	for(int i=1;i<=n;i++)  f[i]=0;
	for(int i=1;i<=n;i++)for(int j=1;j<=n;j++)  g[i][j]=0;
}
	return 0;
}
