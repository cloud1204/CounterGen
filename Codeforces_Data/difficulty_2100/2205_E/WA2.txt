from sys import *
input = stdin.readline
inputlist = lambda: map(int, stdin.readline().split())

def solve():
    n = int(input())
    a = [None] + list(inputlist())
    dp = [1]
    for i in range(1, n+1):
        total = sum(dp)
        current = i
        for j in reversed(range(1, i)):
            if a[j] != a[current]:
                current = i
            if a[j] == a[current]:
                total -= dp[j-1]
                current -= 1
        dp.append(total)
    return dp[-1]


for _ in range(int(input())):
    print(solve())
