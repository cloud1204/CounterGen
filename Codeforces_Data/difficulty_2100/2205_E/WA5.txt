#include<bits/stdc++.h>
using namespace std;
#define int long long
const int N=8e3+66,mod=1e10;
int dp[N],a[N],h[N][N];
int max(int t1,int t2){
	if(t1>t2)return t1;
	return t2;
}
signed main(){
	ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
	int T;cin>>T;
	for(int e=1;e<=T;e++){
		int n;cin>>n;
		for(int i=1;i<=n;i++){
			cin>>a[i];dp[i]=0;
			for(int j=1;j<=n;j++){
				h[i][j]=0;
			}
		}dp[n]=1;dp[n+1]=1;
		for(int i=n-1;i>=1;i--){
			dp[i]+=dp[i+1];
			for(int j=i+1;j<=n;j++){
				if(h[i+1][j]<(j-i+1)/2&&a[i]==a[j-h[i+1][j]]){
					h[i][j]=h[i+1][j]+1;
				}
				if(a[i]==a[j]){
					h[i][j]=1;
				}
				if(h[i][j]==0)dp[i]+=dp[j+1];
				dp[i]%=mod;
			}
		}
		cout<<dp[1]<<endl;
	}
}
